%  写作提示（写作手必读，不要删除本块）

%     表：\tabcaption{表题}{tab:label}{ccc}{表头}{表体}
%  4. 图表后至少 100 字分析；正文不出现 reports/、code/、results.json 等内部文件名。
%  5. 编译：xelatex 两遍；插图必须用 xelatex。
%  6. 本模板已加载 graphicx/float/caption/longtable，不要重复加载。
% =============================================================================
\documentclass[fontset=windows, 12pt, a4paper]{ctexart}

% ---------- 页面：A4，上下左右 2.5cm ----------
\usepackage[a4paper, top=2.5cm, bottom=2.5cm, left=2.5cm, right=2.5cm]{geometry}

% ---------- 数学 ----------
\usepackage{amsmath}
\usepackage{amssymb}

% ---------- 图、表、浮动体 ----------
\usepackage{graphicx}
\usepackage{float}
\usepackage{caption}
\usepackage{longtable}
\usepackage{placeins}
\captionsetup{font=small,labelsep=quad}

% ---------- 字体（fontspec + ctex，xelatex 编译） ----------
\IfFontExistsTF{Times New Roman}{\setmainfont{Times New Roman}}{}
\IfFontExistsTF{Menlo}{\setmonofont{Menlo}}{}

% ---------- 正文：12pt，首行缩进 2em，两端对齐，行距 0.78em ----------
% 0.78em 行距 => 行高 = 1.78 * 字号，\linespread 因子 ≈ 1.483
\linespread{1.483}
\setlength{\parindent}{2em}

% ---------- 标题编号（\ctexset 配置） ----------
% 默认模板使用阿拉伯数字编号：1 / 1.1 / 1.1.1
\ctexset{
  section/number       = \arabic{section},
  subsection/number    = \arabic{section}.\arabic{subsection},
  subsubsection/number = \arabic{section}.\arabic{subsection}.\arabic{subsubsection},
}

% ---------- 标题格式（titlesec 配置） ----------
% 对应 Typst：一级 16pt 居中黑体加粗，二级 14pt 黑体加粗，三级 12pt 黑体加粗
\usepackage{titlesec}
\titleformat{\section}
  {\centering\fontsize{16pt}{19.2pt}\heiti\bfseries}
  {\arabic{section}}{1em}{}
\titleformat{\subsection}
  {\fontsize{14pt}{16.8pt}\heiti\bfseries}
  {\arabic{section}.\arabic{subsection}}{1em}{}
\titleformat{\subsubsection}
  {\fontsize{12pt}{14.4pt}\heiti\bfseries}
  {\arabic{section}.\arabic{subsection}.\arabic{subsubsection}}{1em}{}
\titlespacing{\section}       {0pt}{1.8em}{1.0em}
\titlespacing{\subsection}    {0pt}{1.25em}{0.65em}
\titlespacing{\subsubsection} {0pt}{0.9em}{0.4em}

% ---------- 目录（tocloft 点线引导） ----------
\usepackage{tocloft}
\renewcommand{\cftsecfont}{\heiti\bfseries}
\renewcommand{\cftsecpagefont}{\heiti\bfseries}
\renewcommand{\cftsecleader}{\cftdotfill{\cftdotsep}}
\renewcommand{\cftsubsecleader}{\cftdotfill{\cftdotsep}}
\renewcommand{\cftsubsubsecleader}{\cftdotfill{\cftdotsep}}

% ---------- 表格与代码 ----------
\usepackage{booktabs}
\usepackage{array}

\usepackage{listings}

  basicstyle=\ttfamily\small,

}

% ---------- 页眉页脚（fancyhdr） ----------
% 默认只保留页脚页码：页眉横幅 + 横线属于装饰，多数赛道不需要，评委也不加分。
% 赛事明确要求页眉时，把下面两行的注释打开即可：

\fancyhf{}
\fancyfoot[C]{\thepage}
\renewcommand{\headrulewidth}{0pt}

% ---------- 自定义命令（对应 Typst 函数） ----------
% \papertitle{标题}：居中 22pt 加粗标题
\newcommand{\papertitle}[1]{%
  \thispagestyle{empty}%
  \begin{center}
    {\fontsize{22pt}{26.4pt}\heiti\bfseries #1}
  \end{center}
  \vspace{1em}
}
% \abstractcn{摘要正文}{关键词}：居中“摘  要”标题（14pt 加粗）+ 正文 + “关键词：”加粗 + 关键词 + 换页
\newcommand{\abstractcn}[2]{%
  \begin{center}{\fontsize{14pt}{16.8pt}\heiti\bfseries 摘\hspace{2em}要}\end{center}
  \vspace{0.6em}
  #1\par
  \vspace{0.65em}
  \noindent{\heiti\bfseries 关键词：}#2\par
  \newpage
}
% \tocpage：目录页，标题“目录”居中 16pt 黑体加粗，点线引导，页码从 1 开始
\makeatletter
\newcommand{\tocpage}{%
  \setcounter{page}{1}%
  \begin{center}
    {\fontsize{16pt}{19.2pt}\heiti\bfseries 目录}
  \end{center}
  \vspace{1em}
  \@starttoc{toc}%
  \newpage
}
\makeatother
% \referencescn：参考文献节，标题“参考文献”（不带编号但进目录）
\newcommand{\referencescn}{%
  \addcontentsline{toc}{section}{参考文献}%
  \input{references}%
}
% \appendixcn：附录节，标题“附录 A  核心代码”
\newcommand{\appendixcn}{%
  \section*{附录 A\quad 核心代码}%
  \addcontentsline{toc}{section}{附录 A\quad 核心代码}%
  \input{sections/A_code}%
}
% \threelinetable{标题}{列定义}{表头}{表格内容}：三线表，标题在表格上方居中 10.5pt 黑体加粗
% 调用示例：\threelinetable{表 1 示例}{ccc}{A & B & C}{1 & 2 & 3 \\ 4 & 5 & 6}
\newcommand{\threelinetable}[4]{%
  \begin{center}
    {\fontsize{10.5pt}{12.6pt}\heiti\bfseries #1}\par
    \vspace{0.6em}
    \begin{tabular}{#2}
      \toprule
      #3 \\
      \midrule
      #4 \\
      \bottomrule
    \end{tabular}
  \end{center}
}
% \fig{宽度比例}{图片路径}{图题}{label}：带自动编号的插图
% 示例：\fig{0.85}{../figures/data_q1_convergence.pdf}{问题一收敛曲线}{fig:q1}
\newcommand{\fig}[4]{%
  \begin{figure}[H]
    \centering
    \includegraphics[width=#1\textwidth]{\detokenize{#2}}
    \caption{#3}
    \label{#4}
  \end{figure}
}
% \tabcaption{表题}{label}{列格式}{表头}{表体}：带自动编号的三线表
% 示例：\tabcaption{符号说明}{tab:symbols}{lll}{符号 & 含义 & 单位}{$x_1$ & 固定酸度 & g/L}
\newcommand{\tabcaption}[5]{%
  \begin{table}[H]
    \centering
    \caption{#1}
    \label{#2}
    \begin{tabular}{#3}
      \toprule
      #4 \\
      \midrule
      #5 \\
      \bottomrule
    \end{tabular}
  \end{table}
}
